cmake_minimum_required(VERSION 3.16)
project(taigi-keyboard-linux VERSION 0.1.0 LANGUAGES CXX)

find_package(PkgConfig REQUIRED)
pkg_check_modules(FCITX5 REQUIRED IMPORTED_TARGET Fcitx5Core>=5.1.0)
include(GNUInstallDirs)

set(RUST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../engine")
set(RUST_LIB "${RUST_DIR}/target/release/liblinux_ffi.so")

add_library(taigi MODULE taigi.cpp)
target_compile_features(taigi PRIVATE cxx_std_20)
target_compile_definitions(taigi PRIVATE TAIGI_DICT_DIR="${CMAKE_INSTALL_PREFIX}/share/taigi-keyboard/dictionaries")
target_link_libraries(taigi PRIVATE PkgConfig::FCITX5 "${RUST_LIB}")
set_target_properties(taigi PROPERTIES PREFIX "lib" BUILD_RPATH "${RUST_DIR}/target/release" INSTALL_RPATH "$ORIGIN/../taigi-keyboard")
install(TARGETS taigi DESTINATION "${CMAKE_INSTALL_LIBDIR}/fcitx5")
install(FILES "${RUST_LIB}" DESTINATION "${CMAKE_INSTALL_LIBDIR}/taigi-keyboard")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/taigi-addon.conf" DESTINATION "share/fcitx5/addon" RENAME "taigi.conf")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/taigi.conf" DESTINATION "share/fcitx5/inputmethod")
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../dictionaries/" DESTINATION "share/taigi-keyboard/dictionaries")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../dictionary/LICENSE" DESTINATION "share/doc/taigi-keyboard")
install(PROGRAMS "${CMAKE_CURRENT_SOURCE_DIR}/../ibus/ibus-taigi" DESTINATION "libexec" RENAME "ibus-taigi")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../ibus/online_dictionary.py" DESTINATION "libexec")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../ibus/taigi.xml" DESTINATION "share/ibus/component")
